static string ProbeResource(Type type)
{
    try
    {
        var value = new ResourceManager(type).GetString("Auth_InvalidCredentials");
        return value is null ? "<null>" : value;
    }
    catch (Exception ex)
    {
        return $"ERROR {ex.GetType().Name}: {ex.Message}";
    }
}

static double Percentile(List<double> values, int percentile)
{
    var sorted = values.OrderBy(v => v).ToList();
    var index = (int)Math.Ceiling(percentile / 100.0 * sorted.Count) - 1;
    return sorted[Math.Clamp(index, 0, sorted.Count - 1)];
}

static async Task<ConcurrencyResult> RunConcurrencyLevel(int concurrency, string dbPath)
{
    DeleteDb(dbPath);
    var connectionString = $"Data Source={dbPath};Foreign Keys=False";

    await using (var bootstrap = new SqliteConnection(connectionString))
    {
        await bootstrap.OpenAsync();
        await using var command = bootstrap.CreateCommand();
        command.CommandText = "PRAGMA journal_mode=WAL;";
        await command.ExecuteNonQueryAsync();
    }

    var factory = MakeFactory(connectionString);
    await using (var db = factory.CreateDbContext())
    {
        await db.Database.EnsureCreatedAsync();
        db.SchoolYears.Add(new SchoolYear { Id = 1, Name = "Test Year", Status = "ACTIVE" });
        db.ClassOfferings.Add(new ClassOffering { Id = 1, SchoolYearId = 1, SectionId = 1, Status = "ACTIVE" });
        db.AttendanceSessions.Add(new AttendanceSession
        {
            Id = 1,
            ClassOfferingId = 1,
            SessionDate = DateTime.UtcNow,
            Status = "ACTIVE",
            CreatedAt = DateTime.UtcNow,
            UpdatedAt = DateTime.UtcNow
        });
        db.ClassSchedules.Add(new ClassSchedule
        {
            Id = 1,
            ClassOfferingId = 1,
            DayOfWeek = (byte)(((int)DateTime.Now.DayOfWeek + 6) % 7 + 1),
            StartTime = TimeOnly.FromDateTime(DateTime.Now.AddMinutes(-14)),
            EndTime = TimeOnly.FromDateTime(DateTime.Now.AddMinutes(45))
        });

        for (var i = 1; i <= concurrency; i++)
        {
            db.Students.Add(new Student
            {
                Id = i,
                UserId = i,
                Lrn = $"LRN{i}",
                StudentNumber = $"S{i}",
                FirstName = $"Student{i}",
                LastName = "Load",
                CreatedAt = DateTime.UtcNow,
                UpdatedAt = DateTime.UtcNow
            });
            db.Enrollments.Add(new Enrollment { Id = i, StudentId = i, SchoolYearId = 1, SectionId = 1, Status = "ACTIVE" });
            db.QrTokens.Add(new QrToken
            {
                Id = i,
                Token = $"load-{i}",
                ModelType = QrTokenModelType.STUDENT,
                ModelId = i,
                ExpiresAt = DateTime.UtcNow.AddHours(1)
            });
        }
        await db.SaveChangesAsync();
    }

    var service = new TeacherSessionService(factory, new TestAudit());
    var gate = new TaskCompletionSource(TaskCreationOptions.RunContinuationsAsynchronously);
    var tasks = new List<Task<(bool Success, long Ms, string Error)>>(concurrency);

    for (var i = 1; i <= concurrency; i++)
    {
        var student = i;
        tasks.Add(Task.Run(async () =>
        {
            await gate.Task;
            var sw = Stopwatch.StartNew();
            try
            {
                var result = await service.ScanStudentQrAsync(1, 1, $"load-{student}");
                sw.Stop();
                return (result.Success, sw.ElapsedMilliseconds, result.Success ? "" : result.Message);
            }
            catch (Exception ex)
            {
                sw.Stop();
                return (false, sw.ElapsedMilliseconds, $"{ex.GetType().Name}: {ex.Message}");
            }
        }));
    }

    var wall = Stopwatch.StartNew();
    gate.SetResult();
    var outcomes = await Task.WhenAll(tasks);
    wall.Stop();

    await using (var db = factory.CreateDbContext())
    {
        var stored = await db.AttendanceRecords.CountAsync();
        Out($"CONCURRENCY_RECORDS|{concurrency}|stored={stored}");
    }

    SqliteConnection.ClearAllPools();

    var durations = outcomes.Select(o => (double)o.Ms).ToList();
    var errors = outcomes.Where(o => !o.Success)
        .GroupBy(o => o.Error)
        .Select(g => $"{g.Count()}x {Truncate(g.Key, 90)}")
        .ToList();

    return new ConcurrencyResult
    {
        Success = outcomes.Count(o => o.Success),
        Failed = outcomes.Count(o => !o.Success),
        WallMs = wall.ElapsedMilliseconds,
        AvgMs = durations.Average(),
        P95Ms = Percentile(durations, 95),
        MaxMs = durations.Max(),
        Throughput = outcomes.Length / (wall.Elapsed.TotalMilliseconds / 1000.0),
        ErrorSummary = errors.Count == 0 ? "none" : string.Join(" ; ", errors)
    };
}

static async Task RunRecovery(string workDir)
{
    var dbPath = Path.Combine(workDir, "recovery.db");
    DeleteDb(dbPath);
    var connectionString = $"Data Source={dbPath};Foreign Keys=False";
    var factory = MakeFactory(connectionString);

    await using (var db = factory.CreateDbContext())
    {
        await db.Database.EnsureCreatedAsync();
        db.SchoolYears.Add(new SchoolYear { Id = 1, Name = "Test Year", Status = "ACTIVE" });
        db.ClassOfferings.Add(new ClassOffering { Id = 1, SchoolYearId = 1, SectionId = 1, Status = "ACTIVE" });
        db.AttendanceSessions.Add(new AttendanceSession
        {
            Id = 1,
            ClassOfferingId = 1,
            SessionDate = DateTime.UtcNow,
            Status = "ACTIVE",
            CreatedAt = DateTime.UtcNow,
            UpdatedAt = DateTime.UtcNow
        });
        db.ClassSchedules.Add(new ClassSchedule
        {
            Id = 1,
            ClassOfferingId = 1,
            DayOfWeek = (byte)(((int)DateTime.Now.DayOfWeek + 6) % 7 + 1),
            StartTime = TimeOnly.FromDateTime(DateTime.Now.AddMinutes(-14)),
            EndTime = TimeOnly.FromDateTime(DateTime.Now.AddMinutes(45))
        });
        for (var i = 1; i <= 3; i++)
        {
            db.Students.Add(new Student
            {
                Id = i,
                UserId = i,
                Lrn = $"LRN{i}",
                StudentNumber = $"S{i}",
                FirstName = $"Student{i}",
                LastName = "Recovery",
                CreatedAt = DateTime.UtcNow,
                UpdatedAt = DateTime.UtcNow
            });
            db.Enrollments.Add(new Enrollment { Id = i, StudentId = i, SchoolYearId = 1, SectionId = 1, Status = "ACTIVE" });
            db.QrTokens.Add(new QrToken
            {
                Id = i,
                Token = $"rec-{i}",
                ModelType = QrTokenModelType.STUDENT,
                ModelId = i,
                ExpiresAt = DateTime.UtcNow.AddHours(1)
            });
        }
        await db.SaveChangesAsync();
    }

    var scanner = new TeacherSessionService(factory, new TestAudit());

    // Recovery 1: committed records survive a full close of every pooled connection.
    for (var i = 1; i <= 3; i++)
        await scanner.ScanStudentQrAsync(1, 1, $"rec-{i}");
    int beforeRestart = 0;
    await using (var db = factory.CreateDbContext())
        beforeRestart = await db.AttendanceRecords.CountAsync();

    SqliteConnection.ClearAllPools();

    int afterRestart;
    await using (var db = factory.CreateDbContext())
        afterRestart = await db.AttendanceRecords.CountAsync();
    Out($"RECOVERY|restart|expected={beforeRestart}|actual={afterRestart}" +
        $"|result={(afterRestart == beforeRestart ? "recovered" : "data lost")}");

    // Recovery 2: an interrupted transaction must leave no partial row.
    await using (var db = factory.CreateDbContext())
    {
        await db.Database.BeginTransactionAsync();
        db.AttendanceRecords.Add(new AttendanceRecord
        {
            AttendanceSessionId = 1,
            StudentId = 99,
            MarkedByUserId = 1,
            Status = "PRESENT",
            Hash = "interrupted",
            CreatedAt = DateTime.UtcNow,
            UpdatedAt = DateTime.UtcNow
        });
        await db.SaveChangesAsync();
        // Simulated interruption: the context is torn down before COMMIT.
    }

    int afterInterrupt;
    await using (var db = factory.CreateDbContext())
        afterInterrupt = await db.AttendanceRecords.CountAsync();
    Out($"RECOVERY|interrupted-transaction|expected={beforeRestart}|actual={afterInterrupt}" +
        $"|result={(afterInterrupt == beforeRestart ? "rolled back cleanly" : "partial row persisted")}");

    // Recovery 3: normal operation resumes after every database handle is released.
    var resumeReport = await new ReportService(factory).GetReportAsync();
    ScanResult resume;
    try
    {
        resume = await scanner.ScanStudentQrAsync(1, 1, "rec-1");
        Out($"RECOVERY|resume-after-release|read=ok|sessions={resumeReport.Summary.TotalSessions}" +
            $"|scanResponded=yes|scanSuccess={resume.Success}|scanMessage={resume.Message}");
    }
    catch (Exception ex)
    {
        Out($"RECOVERY|resume-after-release|read={(resumeReport.Summary.TotalSessions >= 0 ? "ok" : "failed")}" +
            $"|scanResponded=no|error={ex.GetType().Name}: {ex.Message}");
    }

    int afterResume;
    await using (var db = factory.CreateDbContext())
        afterResume = await db.AttendanceRecords.CountAsync();
    Out($"RECOVERY|final-record-count|expected={beforeRestart}|actual={afterResume}");

    SqliteConnection.ClearAllPools();
}

static void DeleteDb(string path)
{
    SqliteConnection.ClearAllPools();
    foreach (var suffix in new[] { "", "-wal", "-shm" })
    {
        var file = path + suffix;
        if (File.Exists(file))
            File.Delete(file);
    }
}

static string Truncate(string value, int max) =>
    value.Length <= max ? value : value[..max] + "...";

sealed record ScanCheck(string Name, bool ExpectedSuccess, string? ExpectedStatus, ScanResult Actual)
{
    public bool Passed =>
        Actual.Success == ExpectedSuccess &&
        (ExpectedStatus is null || Actual.Status == ExpectedStatus);
}

sealed record LoginCheck(string Name, bool Expected, LoginResponse Actual)
{
    public bool Passed => Actual.Success == Expected;
}

sealed class ConcurrencyResult
{
    public int Success { get; init; }
    public int Failed { get; init; }
    public long WallMs { get; init; }
    public double AvgMs { get; init; }
    public double P95Ms { get; init; }
    public double MaxMs { get; init; }
    public double Throughput { get; init; }
    public string ErrorSummary { get; init; } = "";
}
